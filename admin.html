📄 New Google Sheet Setup

Yeh Apps Script code copy karke Google Sheets mein paste karo taake nai sheet + API ready ho jaye.

1
Naya Google Sheet kholo → Tools → Apps Script
2
Pura purana code delete karo aur neeche diya hua code paste karo
3
Save → Deploy → New Deployment → Web App → "Anyone" select karo → Copy URL
4
Copy kiya hua URL Settings → API URL mein paste karo
// ═══════════════════════════════════════════════ // ADMIN HUB — Google Apps Script Backend // Google Drive + Sheets Integration // ═══════════════════════════════════════════════ const SHEET_NAME = "Videos"; const DRIVE_FOLDER_ID = "YOUR_DRIVE_FOLDER_ID"; // optional function doGet(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME) || SpreadsheetApp.getActiveSpreadsheet().insertSheet(SHEET_NAME); // Auto-create headers if empty if (sheet.getLastRow() === 0) { sheet.appendRow(["Name","Category","URL","Thumbnail","FileId"]); sheet.getRange(1,1,1,5).setFontWeight("bold"); } const rows = sheet.getDataRange().getValues(); const headers = rows[0]; const data = []; for (let i = 1; i < rows.length; i++) { const row = rows[i]; if (!row[0] && !row[2]) continue; data.push({ rowNumber: i + 1, Name: row[0] || "", Category: row[1] || "", URL: row[2] || "#", Thumbnail: row[3] || "", FileId: row[4] || "" }); } return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME) || SpreadsheetApp.getActiveSpreadsheet().insertSheet(SHEET_NAME); const data = JSON.parse(e.postData.contents); // ── DELETE ── if (data.action === "DELETE") { try { if (!data.keepImage && data.rowNumber) { const fileId = sheet.getRange(data.rowNumber, 5).getValue(); if (fileId) { try { DriveApp.getFileById(fileId).setTrashed(true); } catch(ex) {} } } sheet.deleteRow(data.rowNumber); return jsonOut({ status: "DELETED" }); } catch(ex) { return jsonOut({ status: "ERROR", message: ex.message }); } } // ── EDIT ── if (data.action === "EDIT") { try { const row = data.rowNumber; if (data.name) sheet.getRange(row, 1).setValue(data.name); if (data.category) sheet.getRange(row, 2).setValue(data.category); if (data.url) sheet.getRange(row, 3).setValue(data.url); if (data.thumbnail) { const { url: thumbUrl, fileId } = uploadBase64(data.thumbnail); sheet.getRange(row, 4).setValue(thumbUrl); sheet.getRange(row, 5).setValue(fileId); } return jsonOut({ status: "UPDATED" }); } catch(ex) { return jsonOut({ status: "ERROR", message: ex.message }); } } // ── ADD ── try { let thumbUrl = "", fileId = ""; if (data.thumbnail) { const result = uploadBase64(data.thumbnail); thumbUrl = result.url; fileId = result.fileId; } else if (data.existingThumbUrl) { thumbUrl = data.existingThumbUrl; fileId = data.existingFileId || ""; } sheet.appendRow([data.name, data.category, data.url, thumbUrl, fileId]); return jsonOut({ status: "OK", thumbUrl }); } catch(ex) { return jsonOut({ status: "ERROR", message: ex.message }); } } function uploadBase64(base64Data) { const parts = base64Data.split(","); const mimeType = parts[0].match(/:(.*?);/)[1]; const bytes = Utilities.base64Decode(parts[1]); const blob = Utilities.newBlob(bytes, mimeType, "thumb_" + Date.now() + ".jpg"); let folder; try { folder = DriveApp.getFolderById(DRIVE_FOLDER_ID); } catch(e) { folder = DriveApp.getRootFolder(); } const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); const fileId = file.getId(); const url = "https://lh3.googleusercontent.com/d/" + fileId; return { url, fileId }; } function jsonOut(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
