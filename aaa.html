// ── GOOGLE APPS SCRIPT CODE (Paste in Extensions -> Apps Script) ── var SHEET_NAME = "Sheet1"; var CAT_SHEET_NAME = "CategorySheet"; var DRIVE_FOLDER_ID = "ROOT"; // Ya specific folder ID function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(["Unique_Id", "File_Id", "Name", "Category", "URL", "Thumbnail"]); } if (e.parameter.action === "getCats") { return getCategoriesResponse(); } var data = sheet.getDataRange().getValues(); var result = []; for (var i = 1; i < data.length; i++) { result.push({ rowNumber: i - 1, Name: data[i][2], Category: data[i][3], URL: data[i][4], Thumbnail: data[i][5] }); } return jsonOut(result); } function getCategoriesResponse() { var sheet = getOrCreateCatSheet(); var data = sheet.getDataRange().getValues(); var result = []; for (var i = 1; i < data.length; i++) { if (!data[i][0]) continue; result.push({ rowNumber: i + 1, // 1-based actual row number Category: data[i][0] || "", Icon: data[i][1] || "📁", Status: data[i][2] || "Active", VideoCount: data[i][3] || 0, LastUpdated: data[i][4] || "", Timestamp: data[i][5] || "" }); } return jsonOut(result); } function doPost(request) { try { var data = JSON.parse(request.postData.contents); if (data.action === "ADD_CAT") { var catSheet = getOrCreateCatSheet(); var now = new Date().toISOString(); catSheet.appendRow([ data.category || "", data.icon || "📁", data.status || "Active", data.videoCount || 0, now, data.timestamp || Date.now() ]); var newRow = catSheet.getLastRow(); return jsonOut({ status: "OK", rowNumber: newRow }); } if (data.action === "RENAME_CAT") { var catSheet = getOrCreateCatSheet(); catSheet.getRange(data.rowNumber, 1).setValue(data.newCategory); catSheet.getRange(data.rowNumber, 5).setValue(data.lastUpdated || new Date().toISOString()); // Sheet1 mein bhi saare videos ka category rename karo var ss = SpreadsheetApp.getActiveSpreadsheet(); var vidSheet = ss.getSheetByName(SHEET_NAME); if (vidSheet && data.oldCategory) { var vData = vidSheet.getDataRange().getValues(); for (var vi = 1; vi < vData.length; vi++) { if (String(vData[vi][3]).trim().toLowerCase() === String(data.oldCategory).trim().toLowerCase()) { vidSheet.getRange(vi + 1, 4).setValue(data.newCategory); } } } return jsonOut({ status: "UPDATED" }); } if (data.action === "DELETE_CAT") { var catSheet = getOrCreateCatSheet(); catSheet.deleteRow(data.rowNumber); return jsonOut({ status: "DELETED" }); } if (data.action === "UPDATE_COUNT") { var catSheet = getOrCreateCatSheet(); catSheet.getRange(data.rowNumber, 4).setValue(data.videoCount || 0); catSheet.getRange(data.rowNumber, 5).setValue(data.lastUpdated || new Date().toISOString()); return jsonOut({ status: "UPDATED" }); } var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); if (data.action === undefined) { var timestamp = new Date().getTime(); var uniqueId = "vid_" + timestamp; var fileId = ""; var thumbUrl = "https://placehold.co/600x400/2a3555/white?text=No+Image"; if (data.thumbnail && data.thumbnail.includes("base64")) { var folder = DRIVE_FOLDER_ID === "ROOT" ? DriveApp.getRootFolder() : DriveApp.getFolderById(DRIVE_FOLDER_ID); var imageData = data.thumbnail.split(",")[1]; var blob = Utilities.newBlob(Utilities.base64Decode(imageData), "image/png", "thumb_" + timestamp + ".png"); var file = folder.createFile(blob); fileId = file.getId(); thumbUrl = "https://drive.google.com/thumbnail?id=" + fileId + "&sz=w500"; } else if (data.existingThumbUrl) { thumbUrl = data.existingThumbUrl; fileId = data.existingFileId || ""; } sheet.appendRow([uniqueId, fileId, data.name, data.category, data.url, thumbUrl]); updateVideoCountForCat(data.category); return jsonOut({ status: "OK", uniqueId: uniqueId }); } if (data.action === "DELETE") { var row = Number(data.rowNumber) + 2; var cat = sheet.getRange(row, 4).getValue(); if (!data.keepImage) { var oldFileId = sheet.getRange(row, 2).getValue(); if (oldFileId) { try { DriveApp.getFileById(oldFileId).setTrashed(true); } catch(e) {} } } sheet.deleteRow(row); updateVideoCountForCat(cat); return jsonOut({ status: "DELETED" }); } if (data.action === "EDIT") { var row = Number(data.rowNumber) + 2; var oldCat = sheet.getRange(row, 4).getValue(); sheet.getRange(row, 3).setValue(data.name); sheet.getRange(row, 4).setValue(data.category); sheet.getRange(row, 5).setValue(data.url); if (data.thumbnail && data.thumbnail.includes("base64")) { var oldFid = sheet.getRange(row, 2).getValue(); if (oldFid) { try { DriveApp.getFileById(oldFid).setTrashed(true); } catch(e) {} } var folder = DRIVE_FOLDER_ID === "ROOT" ? DriveApp.getRootFolder() : DriveApp.getFolderById(DRIVE_FOLDER_ID); var imageData = data.thumbnail.split(",")[1]; var blob = Utilities.newBlob(Utilities.base64Decode(imageData), "image/png", "thumb_" + new Date().getTime() + ".png"); var file = folder.createFile(blob); sheet.getRange(row, 2).setValue(file.getId()); sheet.getRange(row, 6).setValue("https://drive.google.com/thumbnail?id=" + file.getId() + "&sz=w500"); } if (oldCat !== data.category) updateVideoCountForCat(oldCat); updateVideoCountForCat(data.category); return jsonOut({ status: "UPDATED" }); } } catch(e) { return jsonOut({ status: "ERROR", message: e.toString() }); } } function getOrCreateCatSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(CAT_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(CAT_SHEET_NAME); sheet.appendRow(["Category","Icon","Status","VideoCount","LastUpdated","Timestamp"]); sheet.getRange(1,1,1,6).setFontWeight("bold").setBackground("#1a1a2e").setFontColor("#ffffff"); } return sheet; } function updateVideoCountForCat(catName) { if (!catName) return; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); var catSheet = getOrCreateCatSheet(); if (!sheet) return; var vData = sheet.getDataRange().getValues(); var count = 0; for (var i = 1; i < vData.length; i++) { if (vData[i][3] === catName) count++; } var cData = catSheet.getDataRange().getValues(); for (var j = 1; j < cData.length; j++) { if (cData[j][0] === catName) { catSheet.getRange(j + 1, 4).setValue(count); catSheet.getRange(j + 1, 5).setValue(new Date().toISOString()); return; } } } function jsonOut(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
